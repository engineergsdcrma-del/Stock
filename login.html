async function login(){ const user = document.getElementById("user").value.trim(); const pass = document.getElementById("pass").value.trim(); if(user==="" || pass===""){ alert("กรุณากรอก Username และ Password"); return; } try{ const res = await fetch(API,{ method:"POST", body:JSON.stringify({ action:"login", username:user, password:pass }) }); const result = (await res.text()).trim(); if(result==="OK"){ localStorage.clear(); localStorage.setItem( "auth", JSON.stringify({ login:true, user:user }) ); location.href="index.html"; }else{ localStorage.clear(); alert("Username หรือ Password ไม่ถูกต้อง"); } }catch(err){ alert("เชื่อมต่อระบบไม่ได้"); } }
